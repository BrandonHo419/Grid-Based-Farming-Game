cmake_minimum_required(VERSION 3.25)
project(GridBasedFarmingGame VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(GRIDFARM_BUILD_APP "Build the graphical application" ON)
include(CTest)
add_library(farm_core src/core/World.cpp src/core/SaveCodec.cpp)
add_library(farm_storage src/storage/SaveStore.cpp)
target_link_libraries(farm_storage PUBLIC farm_core)
target_include_directories(farm_core PUBLIC src)
if(BUILD_TESTING AND NOT EMSCRIPTEN)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE farm_core)
    add_test(NAME core_rules COMMAND core_tests)
    add_executable(save_tests tests/save_tests.cpp)
    target_link_libraries(save_tests PRIVATE farm_storage)
    add_test(NAME save_rules COMMAND save_tests)
    add_executable(offline_tests tests/offline_tests.cpp)
    target_link_libraries(offline_tests PRIVATE farm_storage)
    add_test(NAME offline_rules COMMAND offline_tests)
    add_executable(ui_tests tests/ui_tests.cpp)
    target_link_libraries(ui_tests PRIVATE farm_core)
    add_test(NAME ui_rules COMMAND ui_tests)
endif()
if(GRIDFARM_BUILD_APP)
    include(FetchContent)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    if(EMSCRIPTEN)
        set(PLATFORM Web CACHE STRING "" FORCE)
    endif()
    FetchContent_Declare(raylib
        GIT_REPOSITORY https://github.com/raysan5/raylib.git
        GIT_TAG dbc56a87da87d973a9c5baa4e7438a9d20121d28)
    FetchContent_MakeAvailable(raylib)
    add_executable(grid_farm src/main.cpp src/app/Game.cpp)
    target_link_libraries(grid_farm PRIVATE farm_core farm_storage raylib)
    # these checks need graphics and audio, so keep them out of headless CTest
    if(BUILD_TESTING AND NOT EMSCRIPTEN)
        add_executable(desktop_checks EXCLUDE_FROM_ALL tests/desktop_checks.cpp src/app/Game.cpp)
        target_compile_definitions(desktop_checks PRIVATE GRIDFARM_RUNTIME_CHECKS)
        target_link_libraries(desktop_checks PRIVATE farm_core farm_storage raylib)
    endif()
    if(EMSCRIPTEN)
        set_target_properties(grid_farm PROPERTIES SUFFIX ".html")
        target_link_options(grid_farm PRIVATE
            "--shell-file=${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html"
            "-sALLOW_MEMORY_GROWTH=1")
        set_property(TARGET grid_farm APPEND PROPERTY LINK_DEPENDS
            "${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html")
    endif()
endif()
